Tablas nuevas

CREATE TABLE Imagen (
    Id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    Nombre NVARCHAR(255) NOT NULL DEFAULT '',
    Descripcion NVARCHAR(1000) NOT NULL DEFAULT '',
    FechaCreacion DATETIME2 NOT NULL DEFAULT GETDATE(),
    Url NVARCHAR(500) NOT NULL DEFAULT '',
    UrlThumb NVARCHAR(500) NOT NULL DEFAULT '',
    Tipo NVARCHAR(50) NOT NULL DEFAULT ''
);
